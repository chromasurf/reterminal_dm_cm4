# Optional Overclocking: Max for cm4
# https://www.jeffgeerling.com/blog/2020/overclocking-raspberry-pi-compute-module-4

#arm_freq=2147
arm_freq=2000

gpu_freq=750

# skip some blocks =============================

camera_auto_detect=0

# If your device does not use external HDMI, skip over
# the search for a display to speed up boot time.
display_auto_detect=1
hdmi_ignore_hotplug:0=1
hdmi_ignore_hotplug:1=1
# The kernel's KMS driver sets the display up; the firmware does not probe EDID for it.
disable_fw_kms_setup=1
#disable_overscan=1
#hdmi_enable_4kp60=1

### reTerminal DM ###############################

dtparam=i2c_arm=on
dtparam=i2c_vc=on
# The GT9271 touch controller sits on i2c_vc (behind the CSI/DSI mux).
# 400 kHz is the datasheet maximum; the firmware default is 100 kHz.
dtparam=i2c_vc_baudrate=400000
dtparam=i2s=on
dtparam=spi=on
dtparam=audio=on
dtparam=ant2=on
# Nothing sits on the DM's PCIe lane; brcm-pcie waited 343 ms for a link before init.
dtparam=pcie=off

enable_uart=1
max_framebuffers=2

# Overlays ============================
# The ramoops overlay works with the pstore driver to preserve crash
# information across reboots in DRAM
dtoverlay=ramoops

# For the CM4 I/O board, USB is disabled by default. This enables it.
dtoverlay=dwc2,dr_mode=host
dtoverlay=vc4-kms-v3d,cma-512

dtoverlay=i2c1,pins_2_3
dtoverlay=i2c3,pins_4_5

dtoverlay=reTerminal-DM
# No tpm-spi-slx9670: the reTerminal DM has no TPM, spi0 CS0 is the MCP2518FD CAN controller.
